JSON.parse(e.postData.contents || '{}') : {}; } catch(_) {} // Secret key 검증 const secret = params.secret || body.secret || ''; if (SECRET_KEY && secret !== SECRET_KEY) { return respond({ error: 'Unauthorized' }); } const action = params.action || body.action; try { switch (action) { // ── Students ── case 'getStudents': return respond(getStudents()); case 'saveStudent': return respond(saveStudent(body.student)); case 'saveStudentsBulk': return respond(saveStudentsBulk(body.students)); case 'deleteStudent': return respond(deleteStudent(body.id)); // ── Teachers ── case 'getTeachers': return respond(getTeachers()); case 'saveTeachersBulk': return respond(saveTeachersBulk(body.teachers)); // ── Attendance ── case 'getAttendance': return respond(getAttendance(params.week || body.week)); case 'saveAttendance': return respond(saveAttendance(body.week, body.records)); // ── Teacher Attendance ── case 'getTeacherAttendance': return respond(getTeacherAttendance(params.week || body.week)); case 'saveTeacherAttendance': return respond(saveTeacherAttendance(body.week, body.records)); // ── Email ── case 'sendOfferingEmail': return respond(sendOfferingEmail(body.week, body.records, body.students)); default: return respond({ error: 'Unknown action: ' + action }); } } catch (err) { return respond({ error: err.message, stack: err.stack }); } } function respond(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ─── STUDENTS ────────────────────────────── const STUDENT_HEADERS = [ 'id','nameKo','nameEn','dob','gender','grade', 'parentName','parentPhone','parentEmail','address', 'allergyYn','allergyDetail','photoConsent','noToys','teacherNote', 'active' ]; function getStudents() { const sheet = getSheet('Students'); ensureHeaders(sheet, STUDENT_HEADERS); return { students: sheetToObjects(sheet) }; } function saveStudent(student) { const sheet = getSheet('Students'); ensureHeaders(sheet, STUDENT_HEADERS); return upsert(sheet, STUDENT_HEADERS, student, 'id'); } function saveStudentsBulk(students) { if (!Array.isArray(students)) return { error: 'Not an array' }; const sheet = getSheet('Students'); ensureHeaders(sheet, STUDENT_HEADERS); students.forEach(s => upsert(sheet, STUDENT_HEADERS, s, 'id')); return { success: true, count: students.length }; } function deleteStudent(id) { return deleteRow(getSheet('Students'), 'id', id); } // ─── TEACHERS ────────────────────────────── const TEACHER_HEADERS = ['id','nameKo','nameEn','role','birthday','phone','active']; function getTeachers() { const sheet = getSheet('Teachers'); ensureHeaders(sheet, TEACHER_HEADERS); return { teachers: sheetToObjects(sheet) }; } function saveTeachersBulk(teachers) { if (!Array.isArray(teachers)) return { error: 'Not an array' }; const sheet = getSheet('Teachers'); ensureHeaders(sheet, TEACHER_HEADERS); teachers.forEach(t => upsert(sheet, TEACHER_HEADERS, t, 'id')); return { success: true, count: teachers.length }; } // ─── ATTENDANCE ──────────────────────────── function getAttendance(week) { const sheet = getSheet('Attendance'); ensureHeaders(sheet, ['week','studentId','present','offering','time']); const rows = sheet.getDataRange().getValues().slice(1); const records = {}; rows.forEach(r => { if (String(r[0]) === String(week) && r[1]) { records[r[1]] = { present: r[2] === true || r[2] === 'TRUE', offering: parseFloat(r[3]) || 0, time: r[4] || null }; } }); return { week, records }; } function saveAttendance(week, records) { const sheet = getSheet('Attendance'); ensureHeaders(sheet, ['week','studentId','present','offering','time']); // delete existing week rows (reverse) const all = sheet.getDataRange().getValues(); for (let i = all.length - 1; i >= 1; i--) { if (String(all[i][0]) === String(week)) sheet.deleteRow(i + 1); } Object.entries(records || {}).forEach(([sid, d]) => { sheet.appendRow([week, sid, !!d.present, parseFloat(d.offering||0).toFixed(2), d.time||'']); }); return { success: true, week, count: Object.keys(records||{}).length }; } // ─── TEACHER ATTENDANCE ──────────────────── function getTeacherAttendance(week) { const sheet = getSheet('TeacherAttendance'); ensureHeaders(sheet, ['week','teacherId','present','time']); const rows = sheet.getDataRange().getValues().slice(1); const records = {}; rows.forEach(r => { if (String(r[0]) === String(week) && r[1]) { records[r[1]] = { present: r[2] === true || r[2] === 'TRUE', time: r[3] || null }; } }); return { week, records }; } function saveTeacherAttendance(week, records) { const sheet = getSheet('TeacherAttendance'); ensureHeaders(sheet, ['week','teacherId','present','time']); const all = sheet.getDataRange().getValues(); for (let i = all.length - 1; i >= 1; i--) { if (String(all[i][0]) === String(week)) sheet.deleteRow(i + 1); } Object.entries(records || {}).forEach(([tid, d]) => { sheet.appendRow([week, tid, !!d.present, d.time||'']); }); return { success: true }; } // ─── EMAIL — Offering 공유 ────────────────── function sendOfferingEmail(week, records, students) { const to = OFFERING_EMAIL; if (!to) return { error: 'OFFERING_EMAIL이 설정되지 않았습니다. Apps Script의 OFFERING_EMAIL 변수에 이메일을 입력하세요.' }; let total = 0; const rows = (students || []).map(s => { const d = (records || {})[s.id]; if (!d || !d.present) return null; const amt = parseFloat(d.offering || 0); total += amt; return ` ${s.id} ${s.nameKo} ${s.nameEn} $${amt.toFixed(2)} ${d.time || ''} `; }).filter(Boolean); const html = `

IN2 Dallas Pre-K — 헌금 보고서

주차: ${week}

발송 시간: ${new Date().toLocaleString('ko-KR', {timeZone:'America/Chicago'})}

${rows.join('')}
ID 한국어 이름 English Name 헌금 시간
합계 $${total.toFixed(2)}

IN2 Dallas Pre-K Check-In 시스템에서 자동 발송됨

`; GmailApp.sendEmail(to, `[IN2 Dallas Pre-K] 헌금 보고서 — ${week}`, '', { htmlBody: html }); return { success: true, to, total: total.toFixed(2), count: rows.length }; } // ─── Utility ─────────────────────────────── function getSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); return ss.getSheetByName(name) || ss.insertSheet(name); } function ensureHeaders(sheet, headers) { const first = sheet.getRange(1, 1, 1, headers.length).getValues()[0]; if (!first[0]) { sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.getRange(1, 1, 1, headers.length) .setFontWeight('bold') .setBackground('#EAF7FF'); sheet.setFrozenRows(1); } } function sheetToObjects(sheet) { const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; const headers = rows[0]; return rows.slice(1) .map(row => { const obj = {}; headers.forEach((h, i) => obj[h] = row[i] === '' ? '' : row[i]); return obj; }) .filter(o => o.id); } function upsert(sheet, headers, obj, keyField) { const rows = sheet.getDataRange().getValues(); const keyIdx = headers.indexOf(keyField); for (let i = 1; i < rows.length; i++) { if (String(rows[i][keyIdx]) === String(obj[keyField])) { sheet.getRange(i + 1, 1, 1, headers.length) .setValues([headers.map(h => obj[h] !== undefined ? obj[h] : '')]); return { success: true, action: 'updated', id: obj[keyField] }; } } sheet.appendRow(headers.map(h => obj[h] !== undefined ? obj[h] : '')); return { success: true, action: 'created', id: obj[keyField] }; } function deleteRow(sheet, keyField, keyVal) { const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const keyIdx = headers.indexOf(keyField); for (let i = rows.length - 1; i >= 1; i--) { if (String(rows[i][keyIdx]) === String(keyVal)) { sheet.deleteRow(i + 1); return { success: true, deleted: keyVal }; } } return { success: false, error: 'Not found: ' + keyVal }; } // ─── 최초 설정 함수 (한 번만 실행) ──────────── function setup() { ['Students','Teachers','Attendance','TeacherAttendance'].forEach(n => getSheet(n)); SpreadsheetApp.getActiveSpreadsheet().toast('시트 설정 완료!', 'IN2 Dallas Pre-K', 5); }
